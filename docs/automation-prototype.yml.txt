name: Sync content (Dropbox, Slack, PubMed)

on:
  schedule:
    # 매일 한국시간 08:00 과 18:00 (UTC 23:00, 09:00)
    - cron: "0 23 * * *"
    - cron: "0 9 * * *"
  workflow_dispatch:

permissions:
  contents: write

jobs:
  sync:
    runs-on: ubuntu-latest
    env:
      ANTHROPIC_API_KEY: ${{ secrets.ANTHROPIC_API_KEY }}
      DROPBOX_APP_KEY: ${{ secrets.DROPBOX_APP_KEY }}
      DROPBOX_APP_SECRET: ${{ secrets.DROPBOX_APP_SECRET }}
      DROPBOX_REFRESH_TOKEN: ${{ secrets.DROPBOX_REFRESH_TOKEN }}
      SLACK_BOT_TOKEN: ${{ secrets.SLACK_BOT_TOKEN }}
      SLACK_CHANNEL_ID: ${{ secrets.SLACK_CHANNEL_ID }}
      SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK_URL }}
      SITE_URL: https://innobdlab.github.io
      SYNC_SUMMARY_FILE: ${{ github.workspace }}/.sync_summary.txt
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install -r scripts/requirements.txt

      - name: Dropbox inbox
        if: env.DROPBOX_REFRESH_TOKEN != ''
        run: python scripts/sync_dropbox.py

      - name: Slack commands
        if: env.SLACK_BOT_TOKEN != ''
        run: python scripts/sync_slack.py

      - name: PubMed
        run: python scripts/sync_pubmed.py

      - name: Commit changes
        id: commit
        run: |
          git config user.name "innobd-bot"
          git config user.email "bot@users.noreply.github.com"
          git add -A src/data public/images scripts/state.json
          if git diff --cached --quiet; then
            echo "changed=false" >> $GITHUB_OUTPUT
          else
            git commit -m "content: automated sync $(date -u +%Y-%m-%d)"
            git push
            echo "changed=true" >> $GITHUB_OUTPUT
          fi

      - name: Notify Slack
        if: steps.commit.outputs.changed == 'true'
        run: python scripts/notify_slack.py
